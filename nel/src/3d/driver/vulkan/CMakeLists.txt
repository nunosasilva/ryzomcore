FIND_PACKAGE(Vulkan REQUIRED)

IF(NOT WIN32)
  FIND_PACKAGE(X11 REQUIRED)
ENDIF()

FILE(GLOB SRC CONFIGURE_DEPENDS *.cpp *.h)

IF(WIN32)
  SET(NLDRV_VK_LIB "nel_drv_vulkan_win")
ELSE()
  SET(NLDRV_VK_LIB "nel_drv_vulkan")
ENDIF()

NL_TARGET_DRIVER(${NLDRV_VK_LIB} ${SRC})
TARGET_INCLUDE_DIRECTORIES(${NLDRV_VK_LIB} PRIVATE ${CMAKE_CURRENT_SOURCE_DIR} /usr/include/glslang)
TARGET_LINK_LIBRARIES(${NLDRV_VK_LIB} nel3d nelmisc Vulkan::Vulkan glslang glslang-default-resource-limits SPIRV SPIRV-Tools-opt SPIRV-Tools pthread)
IF(NOT WIN32)
  TARGET_LINK_LIBRARIES(${NLDRV_VK_LIB} ${X11_LIBRARIES})
  IF(X11_Xrender_FOUND)
    TARGET_INCLUDE_DIRECTORIES(${NLDRV_VK_LIB} PRIVATE ${X11_Xrender_INCLUDE_PATH})
    TARGET_COMPILE_DEFINITIONS(${NLDRV_VK_LIB} PRIVATE HAVE_XRENDER)
    TARGET_LINK_LIBRARIES(${NLDRV_VK_LIB} ${X11_Xrender_LIB})
  ENDIF()
  IF(X11_Xcursor_FOUND)
    TARGET_INCLUDE_DIRECTORIES(${NLDRV_VK_LIB} PRIVATE ${X11_Xcursor_INCLUDE_PATH})
    TARGET_COMPILE_DEFINITIONS(${NLDRV_VK_LIB} PRIVATE HAVE_XCURSOR)
    TARGET_LINK_LIBRARIES(${NLDRV_VK_LIB} ${X11_Xcursor_LIB})
  ENDIF()
ENDIF()
NL_DEFAULT_PROPS(${NLDRV_VK_LIB} "NeL, Driver, Video: Vulkan")
NL_ADD_LIB_SUFFIX(${NLDRV_VK_LIB})
NL_ADD_RUNTIME_FLAGS(${NLDRV_VK_LIB})

IF(WITH_INSTALL_LIBRARIES OR NOT WITH_STATIC_DRIVERS)
  INSTALL(TARGETS ${NLDRV_VK_LIB} EXPORT "${project_export_name}"
          LIBRARY DESTINATION ${NL_DRIVER_PREFIX} ARCHIVE DESTINATION ${NL_LIB_PREFIX}
          RUNTIME DESTINATION ${NL_DRIVER_PREFIX} COMPONENT drivers3d)
ENDIF()
